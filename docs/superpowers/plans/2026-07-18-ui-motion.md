# UI Motion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the UI motion spec (`docs/superpowers/specs/2026-07-18-ui-motion-design.md`): color-only hover/press feedback across all interactive surfaces (including message bubbles), View-Transitions directional screen slides keyed to the `parents.ts` hierarchy, and rise-in/pop-in content entrances — all token-driven with reduced-motion fallbacks.

**Architecture:** Three independent tiers. Tier 1 extends existing components with `hover:`/`active:` classes plus a shared `transition-tint duration-fast ease-out` vocabulary (new Tailwind theme entries mapping to existing tokens). Tier 2 adds `src/nav/transitions.ts` — a pure `navDirection()` (ancestry walk over `parentOf`) and a `useTransitionedLocation()` hook that wraps the router location swap in `document.startViewTransition`, with CSS keyed off `html[data-nav-dir]` animating a named `arcan-pane` snapshot. Tier 3 adds `arcan-rise`/`arcan-pop` keyframes applied by an append-after-mount guard in the conversation container and remount-keyed unread badges.

**Tech Stack:** React 19, react-router-dom 7.15 (declarative `<Routes>`), Tailwind v3, hand-rolled CSS keyframes in `src/styles/tokens.css`, View Transitions API (no new dependencies). Vitest for unit, Playwright for e2e, parity harness must stay 142/142.

**Design decisions already locked (do not relitigate):**
- Color-only feedback: NO scale/lift/glow transforms on hover or press.
- Only OWN message bubbles get hover/press feedback (they're the only interactive ones — `onContext` is gated `isMine` in `detail.tsx`); an affordance on non-interactive bubbles would lie.
- No list stagger. Skeletons/toasts/modals keep their existing animations.
- Desktop slides move only the routed pane (the `arcan-pane` named snapshot); sidebar/tab-bar chrome stays in the root snapshot and never slides.
- Known accepted nit: chrome elements that key off the LIVE location (mobile tab-bar visibility, desktop active-row highlight) update at transition start and cross-fade via the root snapshot rather than sliding. Acceptable; do not build a second location context for them.

  from { opacity: 0; transform: scale(0.5); }

.animate-arcan-pop  { animation: arcan-pop var(--dur-base) var(--ease-out) both; }

/* ---------- Screen-transition motion (View Transitions API) ----------
   useTransitionedLocation (src/nav/transitions.ts) stamps
   <html data-nav-dir="forward|back|fade"> for the duration of the
   transition. The routed pane carries [view-transition-name:arcan-pane]
   (AppShell); chrome (sidebar / tab bar) stays in the root snapshot and
   never slides. Browsers without the API skip all of this (instant swap). */
@keyframes arcan-nav-in-right  { from { transform: translateX(100%); } }
@keyframes arcan-nav-out-left  { to   { transform: translateX(-18%); } }
@keyframes arcan-nav-in-left   { from { transform: translateX(-18%); } }
@keyframes arcan-nav-out-right { to   { transform: translateX(100%); } }
@keyframes arcan-fade-out      { to   { opacity: 0; } }

/* Root snapshots: chrome pixels are near-identical across the swap — keep
   the default cross-fade but shorten it so it never outlasts the pane. */
html[data-nav-dir]::view-transition-old(root),
html[data-nav-dir]::view-transition-new(root) {
  animation-duration: var(--dur-fast);
}

html[data-nav-dir="forward"]::view-transition-old(arcan-pane) {
  animation: arcan-nav-out-left var(--dur-nav) var(--ease-out) both;
}
html[data-nav-dir="forward"]::view-transition-new(arcan-pane) {
  animation: arcan-nav-in-right var(--dur-nav) var(--ease-out) both;
}
html[data-nav-dir="back"]::view-transition-old(arcan-pane) {
  animation: arcan-nav-out-right var(--dur-nav) var(--ease-out) both;
  z-index: 1; /* the departing child rides OVER the returning parent */
}
html[data-nav-dir="back"]::view-transition-new(arcan-pane) {
  animation: arcan-nav-in-left var(--dur-nav) var(--ease-out) both;
}
html[data-nav-dir="fade"]::view-transition-old(arcan-pane) {
  animation: arcan-fade-out var(--dur-fast) var(--ease-in) both;
}
html[data-nav-dir="fade"]::view-transition-new(arcan-pane) {
  animation: arcan-fade-in var(--dur-fast) var(--ease-out) both;
}

// src/nav/transitions.ts — screen-transition direction + the transitioned
// location hook (UI motion spec, docs/superpowers/specs/2026-07-18-ui-motion-design.md).
//
// Direction derives from the SAME hierarchy that drives the header up button
// (parents.ts): navigating to a descendant slides forward, to an ancestor
// slides back; unrelated moves (tab roots, cross-branch jumps, anything
// touching the auth flow) cross-fade.

import { parentOf } from "./parents";

export type NavDirection = "forward" | "back" | "fade";

const AUTH_ROOTS = ["/auth", "/onboarding", "/pair", "/invite"];

function normalize(path: string): string {
  return path.split("?")[0].replace(/\/+$/, "") || "/";
}

function inAuthFlow(path: string): boolean {
  return AUTH_ROOTS.some((r) => path === r || path.startsWith(`${r}/`));
}

/** Walks the parentOf chain upward from `of`, looking for `candidate`. */
function isAncestor(candidate: string, of: string): boolean {
  let cur = of;
  // parents.ts is a finite tree rooted at "/" — 10 hops far exceeds its depth.
  for (let i = 0; i < 10; i++) {
    const parent = normalize(parentOf(cur));
    if (parent === cur) return false; // fixpoint ("/" → "/")
    if (parent === candidate) return true;
    cur = parent;
  }
  return false;
}

export function navDirection(from: string, to: string): NavDirection {
  const a = normalize(from);
  const b = normalize(to);
  if (a === b) return "fade";
  if (inAuthFlow(a) || inAuthFlow(b)) return "fade";
  if (isAncestor(b, a)) return "back";
  if (isAncestor(a, b)) return "forward";
  return "fade";
}
```

// --- appended below navDirection ---

import { useLayoutEffect, useRef, useState } from "react";
import { flushSync } from "react-dom";
import type { Location } from "react-router-dom";

// lib.dom's startViewTransition typing depends on the TS version — a narrow
// local type keeps us independent of it.
type DocumentWithVT = Document & {
  startViewTransition?: (cb: () => void) => { finished: Promise<void> };
};

/**
 * Returns the location the route table should RENDER. When the router
 * location changes, the swap is wrapped in document.startViewTransition with
 * html[data-nav-dir] stamped so tokens.css can animate the old/new
 * arcan-pane snapshots. Browsers without the API — and users with
 * prefers-reduced-motion — get the plain instant swap.
 *
 * Accepted nit (recorded in the plan header): chrome that keys off the LIVE
 * location (tab-bar visibility, active-row highlight) updates at transition
 * start and cross-fades via the root snapshot instead of sliding.
 */
export function useTransitionedLocation(location: Location): Location {
  const [displayed, setDisplayed] = useState(location);
  const displayedRef = useRef(location);

  useLayoutEffect(() => {
    const from = displayedRef.current;
    if (from.key === location.key) return;

    const commit = () => {
      displayedRef.current = location;
      setDisplayed(location);
    };

    const doc = document as DocumentWithVT;
    const reduceMotion = window.matchMedia(
      "(prefers-reduced-motion: reduce)",
    ).matches;
    if (typeof doc.startViewTransition !== "function" || reduceMotion) {
      commit();
      return;
    }

    document.documentElement.dataset.navDir = navDirection(
      from.pathname + from.search,
      location.pathname + location.search,
    );
    const vt = doc.startViewTransition(() => {
      flushSync(commit);
    });
    void vt.finished.finally(() => {
      delete document.documentElement.dataset.navDir;
    });
  }, [location]);

  return displayed;
}
```
